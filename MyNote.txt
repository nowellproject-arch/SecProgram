MonthlyRecord upodater

(async () => {
    const number = "200";
    const activePublishers = "93";

    const db = await openIndexedDB();

    const tx = db.transaction("MonthlyRecords", "readwrite");
    const store = tx.objectStore("MonthlyRecords");

    const req = store.get(number);

    req.onsuccess = () => {
        const record = req.result;

        if (!record) {
            console.error("❌ NUMBER", number, "not found");
            db.close();
            return;
        }

        console.log("Before:", record);

        record.ActivePublishers = activePublishers;

        const putReq = store.put(record);

        putReq.onsuccess = () => {
            console.log(
                "✅ Updated NUMBER",
                number,
                "ActivePublishers =",
                activePublishers
            );
        };

        putReq.onerror = () => {
            console.error("❌ Update failed:", putReq.error);
        };
    };

    req.onerror = () => {
        console.error("❌ Read failed:", req.error);
    };

    tx.oncomplete = () => {
        db.close();
        console.log("✅ Transaction complete");
    };

    tx.onerror = () => {
        console.error("❌ Transaction error:", tx.error);
        db.close();
    };
})();
